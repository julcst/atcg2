\subsection*{a)}
	$q_u(u,v) = (1, 0, 2u), \qquad q_v(u,v) = (0, 1, 4v)$\\
	\\
	$E = q_u\boldsymbol{\cdot} q_u, \quad F = q_u\boldsymbol{\cdot} q_v, \quad G = q_v\boldsymbol{\cdot} q_v$\\
	\\
	$I(u,v)= \left [
	\begin{array}{cc}
		E & F \\
		F & G \\
	\end{array}
	\right ] = 
	\left [
	\begin{array}{cc}
		1 + 4u^2 & 8uv \\
		8uv & 1 + 16v^2 \\
	\end{array}
	\right ] $
	
\subsection*{b)}

	\begin{align*}
		A &= \bigintsss\bigintss_D \sqrt{EG-F^2}\,dudv = \bigintsss\bigintss_D \sqrt{(1 + 4u^2)(1 + 16v^2)-(8uv)^2}\,dudv\\
		  &= \bigintsss\bigintsss_D \sqrt{1 + 4u^2 + 16v^2 + 64u^2v^2 - 64u^2v^2}\,dudv = \bigintsss\bigintsss_D \sqrt{1 + 4u^2 + 16v^2}\,dudv
	\end{align*}
	Numerical Solving using CAS:\\
	\begin{align*}
	A = \bigintsss_0^1\bigintsss_0^1 \sqrt{1 + 4u^2 + 16v^2}\,dudv = 2.61676039985 
	\end{align*}
