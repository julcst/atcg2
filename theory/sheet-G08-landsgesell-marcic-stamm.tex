\subsection*{Assignment 2}
The divergence of a vector field tells how much the field is expanding or compressing at a specific point. In other words: It is a measure for the rate of flow entering or leaving in an infinitesimal neighborhood around a point.\\\\
The divergence is given by: $\divergence \vecarrow{f} = \sum_{i=1}^{n} \frac{\partial f}{\partial x_i}$\\\\
Sign of the divergence for the given vector fields (assuming we are looking at the central region for the first two):
\begin{itemize}
	\item Left: $\divergence \vecarrow{f} > 0$ as $\frac{\partial f}{\partial x} > 0$ and $\frac{\partial f}{\partial y} > 0$, the central point is a "source" and the field is expanding as there is more flowing "out" than "in".
	\item Middle: $\divergence \vecarrow{f} < 0$ as $\frac{\partial f}{\partial x} < 0$ and $\frac{\partial f}{\partial y} < 0$, the central point is a "sink" and the field is compressing as there is more flowing "in" than "out".
	\item Right: $\divergence \vecarrow{f} = 0$ as $\frac{\partial f}{\partial x} = 0$ and $\frac{\partial f}{\partial x} = 0$, no compression or expansion, the same amount flowing "in" and "out".
\end{itemize}

	\gradient  \rho(\vecarrow{x}) = \begin{pmatrix}
		-2x \exp(-x^2-y^2)\\
		-2y \exp(-x^2-y^2)
	\end{pmatrix}
\end{align*}

\begin{align*}
	\vecarrow{f} = \begin{pmatrix}
		2 \lambda x \exp(-x^2-y^2)\\
		2 \lambda y \exp(-x^2-y^2)
	\end{pmatrix}
\end{align*}

\begin{align*}
	\divergence \vecarrow{f} &= \frac{\partial f_x}{\partial x} + \frac{\partial f_y}{\partial y}\\
	&= \frac{\partial}{\partial x} 2 \lambda x \exp(-x^2-y^2) + \frac{\partial}{\partial y} 2 \lambda y \exp(-x^2-y^2)\\
	&= 2 \lambda \exp(-x^2-y^2) (1 - 2x^2) + 2 \lambda \exp(-x^2-y^2) (1 - 2y^2)\\
	&= 4 \lambda \exp(-x^2-y^2) (1 - x^2 - y^2)
\end{align*}

\bigskip
\noindent
Divergence at $\vecarrow{x} = (0,0)^T$:
\begin{align*}
	\divergence \vecarrow{f}(0,0) &= 4 \lambda \exp(-0^2-0^2) (1 - 0^2 - 0^2) = 4 \lambda
\end{align*}

Assuming $\lambda > 0$, the divergence at $\vecarrow{x} = (0,0)^T$ is positive, therefore the temperature field has a heat source at this point.

\subsection*{a)}

\begin{align*}
	\divergence\left(\gradient{f}\right) &= \divergence\left(\begin{pmatrix} \frac{\partial f}{\partial x_1}\\ \vdots \\ \frac{\partial f}{\partial x_n} \end{pmatrix}\right)\\
	&= \sum_{i=1}^{n} \frac{\partial}{\partial x_i} \left(\frac{\partial f}{\partial x_i}\right)\\
	&= \sum_{i=1}^{n} \frac{\partial^2 f}{\partial x_i^2}\\
	&= \laplacian f
\end{align*}

\subsection*{b)}

\begin{align*}
	\laplacian \left(\laplacian f\right) &= \laplacian \left( \frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial y^2}\right)\\
	&= \frac{\partial^2}{\partial x^2} \left( \frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial y^2}\right) + \frac{\partial^2}{\partial y^2} \left( \frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial y^2}\right)\\
	&= \frac{\partial^4 f}{\partial x^4} + \frac{\partial^4 f}{\partial x^2 \partial y^2} + \frac{\partial^4 f}{\partial y^2 \partial x^2} + \frac{\partial^4 f}{\partial y^4}\\
	&= \frac{\partial^4 f}{\partial x^4} + \frac{\partial^4 f}{\partial y^4} + 2 \frac{\partial^4 f}{\partial x^2 \partial y^2}
\end{align*}

\subsection*{c)}
If $f$ minimizes $E$ subject to the boundary conditions, then the directional derivatives $\frac{d}{d\epsilon}E\left[f+\epsilon h\right]\Bigg|_{\epsilon=0} = 0$ for all admissible perturbations $h$.

\begin{align*}
	&E\left[f+\epsilon h\right] = \frac{1}{2} \int \int_{R^2} \left(\frac{\partial^2(f+\epsilon h)}{\partial x^2}\right)^2 + 2\left(\frac{\partial^2(f+\epsilon h)}{\partial x \partial y}\right)^2 + \left(\frac{\partial^2(f+\epsilon h)}{\partial y^2}\right)^2 \, dx\, dy\\
	&= \frac{1}{2} \int \int_{R^2} \left(\frac{\partial^2(f^2 + 2f\epsilon h + \epsilon^2 h^2)}{\partial x^2}\right) + 2\left(\frac{\partial^2(f^2 + 2f\epsilon h + \epsilon^2 h^2)}{\partial x \partial y}\right) + \left(\frac{\partial^2(f^2 + 2f\epsilon h + \epsilon^2 h^2)}{\partial y^2}\right) \, dx\, dy\\
\end{align*}

Differentiating with respect to $\epsilon$:
\begin{align*}
	\frac{d}{d\epsilon}E\left[f+\epsilon h\right] = \frac{1}{2} \int \int_{R^2} \left(\frac{\partial^2(2f h + 2\epsilon h^2)}{\partial x^2}\right) + 2\left(\frac{\partial^2(2f h + 2\epsilon h^2)}{\partial x \partial y}\right) + \left(\frac{\partial^2(2f h + 2\epsilon h^2)}{\partial y^2}\right) \, dx\, dy
\end{align*}
\begin{align*}
	\implies \frac{d}{d\epsilon}E\left[f+\epsilon h\right]\Bigg|_{\epsilon=0} &= \int \int_{R^2} \left(\frac{\partial^2(f h)}{\partial x^2}\right) + 2\left(\frac{\partial^2(f h)}{\partial x \partial y}\right) + \left(\frac{\partial^2(f h)}{\partial y^2}\right) \, dx\, dy\\
	&= \underbrace{\int \int_{R^2} \left(\frac{\partial^2(f h)}{\partial x^2}\right) \, dx\, dy}_{\text{Term 1}} + 2\underbrace{\int \int_{R^2} \left(\frac{\partial^2(f h)}{\partial x \partial y}\right) \, dx\, dy}_{\text{Term 2}} + \underbrace{\int \int_{R^2} \left(\frac{\partial^2(f h)}{\partial y^2}\right) \, dx\, dy}_{\text{Term 3}}\\
\end{align*}
Let us first consider Term 1 and let's apply Green’s first idendity twice:
\begin{align*}
	\text{Term 1} &= \int \int_{R^2} \frac{\partial^2 f}{\partial x^2} \frac{\partial^2 h}{\partial x^2} \, dx\, dy\\
	&= \underbrace{\int \int_{\partial R^2} \frac{\partial h}{\partial x} \left (\frac{\partial^2 f}{\partial x^2} \boldsymbol{\cdot} \tilde{v}\right ) \, dx\, dy}_{0 \text{ due to } \partial R^2 = \emptyset} - \int \int_{R^2} \frac{\partial^3 f}{\partial x^3} \frac{\partial h}{\partial x} \, dx\, dy\\
	- \int \int_{R^2} \frac{\partial^3 f}{\partial x^3} \frac{\partial h}{\partial x} \, dx\, dy &= -\underbrace{\int \int_{\partial R^2} h \left (\frac{\partial^3 f}{\partial x^3} \boldsymbol{\cdot} \tilde{v}\right ) \, dx\, dy}_{0 \text{ due to } \partial R^2 = \emptyset} + \int \int_{R^2} \frac{\partial^4 f}{\partial x^4} h \, dx\, dy
\end{align*}
So we end up with:
\begin{align*}
	\int \int_{R^2} \frac{\partial^2 f}{\partial x^2} \frac{\partial^2 h}{\partial x^2} \, dx\, dy = \int \int_{R^2} \frac{\partial^4 f}{\partial x^4} h \, dx\, dy
\end{align*}
Applying the same steps to Term 3 yields:
\begin{align*}
	\int \int_{R^2} \frac{\partial^2 f}{\partial y^2} \frac{\partial^2 h}{\partial y^2} \, dx\, dy = \int \int_{R^2} \frac{\partial^4 f}{\partial y^4} h \, dx\, dy
\end{align*}
And for Term 2 we get:
\begin{align*}
	2\int \int_{R^2} \frac{\partial^2 f}{\partial x \partial y} \frac{\partial^2 h}{\partial x \partial y} \, dx\, dy = 2\int \int_{R^2} \frac{\partial^4 f}{\partial x^2 \partial y^2} h \, dx\, dy
\end{align*}
Now let's put everything together again:
\begin{align*}
	\frac{d}{d\epsilon}E\left[f+\epsilon h\right] &= \int \int_{R^2} \left(\frac{\partial^2(f h)}{\partial x^2}\right) \, dx\, dy + 2\int \int_{R^2} \left(\frac{\partial^2(f h)}{\partial x \partial y}\right) \, dx\, dy + \int \int_{R^2} \left(\frac{\partial^2(f h)}{\partial y^2}\right) \, dx\, dy\\
	&= \int \int_{R^2} \underbrace{\left(\frac{\partial^4 f}{\partial x^4} + 2 \frac{\partial^4 f}{\partial x^2 \partial y^2} + \frac{\partial^4 f}{\partial y^4} \right)}_{\Delta^2 f \text{ we know that from b)}} h \, dx\, dy\\
	&= \int \int_{R^2} (\Delta^2 f) \, h \, dx\, dy
\end{align*}
Since $\frac{d}{d\epsilon}E\left[f+\epsilon h\right]\Bigg|_{\epsilon=0} = 0$ for all admissible perturbations $h$, f minimizes E.\\
It follows that $\Delta^2 f = 0$, because h is arbitrary. $\square$
